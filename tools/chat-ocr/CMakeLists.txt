cmake_minimum_required(VERSION 3.17)
project(chat_ocr LANGUAGES CXX)
# RAPID_SOURCE points at the expanded Project_RapidOcrNcnn-1.2.0 release.
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(OpenCV_STATIC ON)
set(OpenCV_DIR "${RAPID_SOURCE}/opencv-static/windows-x64/x64/vc16/staticlib")
set(ncnn_DIR "${RAPID_SOURCE}/ncnn-vulkan-static/windows-x64/lib/cmake/ncnn")
find_package(ncnn REQUIRED)
find_package(OpenCV REQUIRED)
set(sources AngleNet clipper CrnnNet DbNet OcrLite OcrUtils OcrResultUtils)
list(TRANSFORM sources PREPEND "${RAPID_SOURCE}/src/")
list(TRANSFORM sources APPEND ".cpp")
add_executable(chat-ocr worker.cpp ${sources})
target_include_directories(chat-ocr PRIVATE "${RAPID_SOURCE}/include" "${RAPID_SOURCE}/ncnn-vulkan-static/windows-x64/include")
target_compile_definitions(chat-ocr PRIVATE __VULKAN__ NOMINMAX)
target_link_libraries(chat-ocr PRIVATE ncnn ${OpenCV_LIBS})
